[PDO_TEST]
// Test result variable
:define:var test_res U8 0
// Define RPDO
:define:var obj6200_1 U8 0
:define:var obj6200_2 U8 0
:define:var obj6200_3 U8 0
:define:var obj6200_4 U8 0
:define:can_msg RPDO1 0x200 + nodeid 4
:assign:vars2can_msg RPDO1 obj6200_1 obj6200_2 obj6200_3 obj6200_4
// Update variables
:compute:var obj6200_1 = obj6200_1 | 0x1F?
:compute:var obj6200_3 = obj6200_3 | 0x80?
:compute:var obj6200_4 = obj6200_4 | 0x02?
// Define TPDO
:define:var obj6000_1 U8 0
:define:var obj6000_2 U8 0
:define:var obj6000_3 U8 0
:define:var obj6000_4 U8 0
:define:var obj6000_5 U8 0
:define:can_msg TPDO1 0x180 + nodeid 5
:assign:vars2can_msg TPDO1 obj6000_1 obj6000_2 obj6000_3 obj6000_4 obj6000_5
// Send PDO
:send:can_msg RPDO1
:wait:time 40
// Analyze output
:compute:var obj6000_1 = obj6000_1 & 0x7c?
:compute:var obj6000_5 = obj6000_5 & 0xef?
:compute:var obj6000_3 = obj6000_3 & 0x02?
do_if expr obj6000_1 != 0x7c?
	:read:set:var test_res 1?
done_if
do_if expr obj6000_5 == 0?
	:read:set:var test_res 1?
done_if
do_if expr obj6000_3 != 0?
	:read:set:var test_res 1?
done_if
// Update variables
:compute:var obj6200_1 = obj6200_1 & 0xE0?
:compute:var obj6200_3 = obj6200_3 & 0x7F?
:compute:var obj6200_4 = obj6200_4 & 0xFD?
:send:can_msg RPDO1
:wait:time 500
// Analyze output
:compute:var obj6000_1 = obj6000_1 & 0x7c?
:compute:var obj6000_5 = obj6000_5 & 0x10?
:compute:var obj6000_3 = obj6000_3 & 0xfd?
do_if expr obj6000_1 != 0?
	:read:set:var test_res 1?
done_if
do_if expr obj6000_5 != 0?
	:read:set:var test_res 1?
done_if
do_if expr obj6000_3 == 0?
	:read:set:var test_res 1?
done_if
:print:var test_res
do_if test_res == 0?
	:print:string Test successful
do_else?
	:print:string Test failed
done_if
